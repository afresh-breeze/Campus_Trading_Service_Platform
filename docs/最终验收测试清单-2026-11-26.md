### A.5 关键 SQL 速查

```sql
-- 订单与日志
SELECT * FROM errand_orders ORDER BY order_id DESC LIMIT 10;
SELECT * FROM errand_status_logs WHERE order_id = ? ORDER BY log_id;

-- 会话与消息
SELECT * FROM conversations ORDER BY conversation_id DESC LIMIT 10;
SELECT * FROM messages WHERE conversation_id = ? ORDER BY message_id;

-- 收藏与商品
SELECT favorite_count FROM items WHERE item_id = ?;
SELECT * FROM favorites WHERE item_id = ?;
SELECT * FROM item_status_logs WHERE item_id = ? ORDER BY log_id;

-- 信用分/完成数
SELECT u.student_no, u.credit_score, rp.completed_count
FROM users u LEFT JOIN runner_profiles rp ON rp.user_id = u.user_id;
```
